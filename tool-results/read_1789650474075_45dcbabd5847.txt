   219→   219→function firstString(record: Record<string, unknown>, keys: string[]): string {
   220→   220→  for (const key of keys) {
   221→   221→    const value = record[key];
   222→   222→    if (typeof value === "string" && value.trim().length > 0) return value.trim();
   223→   223→    if (typeof value === "number") return String(value);
   224→   224→  }
   225→   225→  return "";
   226→   226→}

   228→   228→/** Normalise une liste de référentiel (rubriques, pages, dossiers, tags). */
   229→   229→function toRefList(raw: unknown, kind: RefKind): RefItem[] {
   230→   230→  if (!Array.isArray(raw)) return [];
   231→   231→  const out: RefItem[] = [];
   232→   232→  for (const entry of raw) {
   233→   233→    if (typeof entry !== "object" || entry === null) continue;
   234→   234→    const record = entry as Record<string, unknown>;
   235→   235→    const id = firstString(record, ["id", "key", "slug", "code"]);
   236→   236→    if (!id) continue;
   237→   237→    const label = firstString(
   238→   238→      record,
   239→   239→      kind === "pages" ? ["title", "name", "label"] : ["name", "title", "label"]
   240→   240→    );
   241→   241→    if (!label) continue;
   242→   242→    const slug = firstString(record, ["slug"]);
   243→   243→    out.push({
   244→   244→      id,
   245→   245→      label,
   246→   246→      hint: slug && slug !== label ? slug : undefined,
   247→   247→    });
   248→   248→  }
   249→   249→  return out;
   250→   250→}

   252→   252→const positionOf = (item: MenuItem): number =>
   253→   253→  Number.isFinite(item.position) ? item.position : 0;
   254→   254→
   255→   255→/**
   256→   256→ * Remet l'arbre à plat dans un ordre canonique : racines triées par
   257→   257→ * position puis enfants triés par position ; les orphelins (parent supprimé)
   258→   258→ * sont remontés à la racine ; les positions sont renumérotées par groupe.
   259→   259→ */
   260→   260→function normalizeItems(raw: MenuItem[]): MenuItem[] {
   261→   261→  const ids = new Set(raw.map((item) => item.id));
   262→   262→  const roots = raw
   263→   263→    .filter((item) => !item.parent_id || !ids.has(item.parent_id))
   264→   264→    .sort((a, b) => positionOf(a) - positionOf(b));
   265→   265→
   266→   266→  const seen = new Set<string>();
   267→   267→  const out: MenuItem[] = [];
   268→   268→  roots.forEach((root, ri) => {
   269→   269→    seen.add(root.id);
   270→   270→    out.push({ ...root, parent_id: null, position: ri });
   271→   271→    raw
   272→   272→      .filter((item) => item.parent_id === root.id)
   273→   273→      .sort((a, b) => positionOf(a) - positionOf(b))
   274→   274→      .forEach((child, ci) => {
   275→   275→        seen.add(child.id);
   276→   276→        out.push({ ...child, position: ci });
   277→   277→      });
   278→   278→  });
   279→   279→  raw
   280→   280→    .filter((item) => !seen.has(item.id))
   281→   281→    .sort((a, b) => positionOf(a) - positionOf(b))
   282→   282→    .forEach((lost, li) => {
   283→   283→      out.push({ ...lost, parent_id: null, position: roots.length + li });
   284→   284→    });
   285→   285→  return out;
   286→   286→}

   288→   288→/** Aplatit pour l'affichage avec index/compte de fratrie (flèches). */
   289→   289→function flattenItems(items: MenuItem[]): FlatRow[] {
   290→   290→  const normalized = normalizeItems(items);
   291→   291→  const roots = normalized.filter((item) => !item.parent_id);
   292→   292→  const rows: FlatRow[] = [];
   293→   293→  roots.forEach((root, ri) => {
   294→   294→    rows.push({ item: root, depth: 0, sibIndex: ri, sibCount: roots.length });
   295→   295→    const children = normalized.filter((item) => item.parent_id === root.id);
   296→   296→    children.forEach((child, ci) => {
   297→   297→      rows.push({
   298→   298→        item: child,
   299→   299→        depth: 1,
   300→   300→        sibIndex: ci,
   301→   301→        sibCount: children.length,
   302→   302→      });
   303→   303→    });
   304→   304→  });
   305→   305→  return rows;
   306→   306→}

   603→   603→  function moveItem(id: string, direction: -1 | 1) {
   604→   604→    setItems((prev) => {
   605→   605→      const target = prev.find((item) => item.id === id);
   606→   606→      if (!target) return prev;
   607→   607→      const siblings = prev
   608→   608→        .filter((item) => item.parent_id === target.parent_id)
   609→   609→        .sort((a, b) => positionOf(a) - positionOf(b));
   610→   610→      const index = siblings.findIndex((item) => item.id === id);
   611→   611→      const other = siblings[index + direction];
   612→   612→      if (!other) return prev;
   613→   613→      return normalizeItems(
   614→   614→        prev.map((item) => {
   615→   615→          if (item.id === id) return { ...item, position: other.position };
   616→   616→          if (item.id === other.id) return { ...item, position: target.position };
   617→   617→          return item;
   618→   618→        })
   619→   619→      );
   620→   620→    });
   621→   621→  }
