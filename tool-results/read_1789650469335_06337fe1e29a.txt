   212→function genId(): string {
   213→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {

   216→  return `tmp-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;

   219→function firstString(record: Record<string, unknown>, keys: string[]): string {
   220→  for (const key of keys) {
   221→    const value = record[key];
   222→    if (typeof value === "string" && value.trim().length > 0) return value.trim();
   223→    if (typeof value === "number") return String(value);
   224→  }
   225→  return "";
   226→}

   228→/** Normalise une liste de référentiel (rubriques, pages, dossiers, tags). */
   229→function toRefList(raw: unknown, kind: RefKind): RefItem[] {
   230→  if (!Array.isArray(raw)) return [];
   231→  const out: RefItem[] = [];
   232→  for (const entry of raw) {
   233→    if (typeof entry !== "object" || entry === null) continue;
   234→    const record = entry as Record<string, unknown>;
   235→    const id = firstString(record, ["id", "key", "slug", "code"]);
   236→    if (!id) continue;
   237→    const label = firstString(
   238→      record,
   239→      kind === "pages" ? ["title", "name", "label"] : ["name", "title", "label"]
   240→    );
   241→    if (!label) continue;
   242→    const slug = firstString(record, ["slug"]);
   243→    out.push({
   244→      id,
   245→      label,
   246→      hint: slug && slug !== label ? slug : undefined,
   247→    });
   248→  }
   249→  return out;
   250→}

   252→const positionOf = (item: MenuItem): number =>
   253→  Number.isFinite(item.position) ? item.position : 0;
   254→
   255→/**
   256→ * Remet l'arbre à plat dans un ordre canonique : racines triées par
   257→ * position puis enfants triés par position ; les orphelins (parent supprimé)
   258→ * sont remontés à la racine ; les positions sont renumérotées par groupe.
   259→ */
   260→function normalizeItems(raw: MenuItem[]): MenuItem[] {
   261→  const ids = new Set(raw.map((item) => item.id));
   262→  const roots = raw
   263→    .filter((item) => !item.parent_id || !ids.has(item.parent_id))
   264→    .sort((a, b) => positionOf(a) - positionOf(b));
   265→
   266→  const seen = new Set<string>();
   267→  const out: MenuItem[] = [];
   268→  roots.forEach((root, ri) => {
   269→    seen.add(root.id);
   270→    out.push({ ...root, parent_id: null, position: ri });
   271→    raw
   272→      .filter((item) => item.parent_id === root.id)
   273→      .sort((a, b) => positionOf(a) - positionOf(b))
   274→      .forEach((child, ci) => {
   275→        seen.add(child.id);
   276→        out.push({ ...child, position: ci });
   277→      });
   278→  });
   279→  raw
   280→    .filter((item) => !seen.has(item.id))
   281→    .sort((a, b) => positionOf(a) - positionOf(b))
   282→    .forEach((lost, li) => {
   283→      out.push({ ...lost, parent_id: null, position: roots.length + li });
   284→    });
   285→  return out;
   286→}

   288→/** Aplatit pour l'affichage avec index/compte de fratrie (flèches). */
   289→function flattenItems(items: MenuItem[]): FlatRow[] {
   290→  const normalized = normalizeItems(items);
   291→  const roots = normalized.filter((item) => !item.parent_id);
   292→  const rows: FlatRow[] = [];
   293→  roots.forEach((root, ri) => {
   294→    rows.push({ item: root, depth: 0, sibIndex: ri, sibCount: roots.length });
   295→    const children = normalized.filter((item) => item.parent_id === root.id);
   296→    children.forEach((child, ci) => {
   297→      rows.push({
   298→        item: child,
   299→        depth: 1,
   300→        sibIndex: ci,
   301→        sibCount: children.length,
   302→      });
   303→    });
   304→  });
   305→  return rows;
   306→}

   318→    highlight: item.highlight,

   320→    is_visible: item.is_visible,
   321→  }));
   322→}

   591→            item.id === dialogItemId ? { ...item, ...common } : item

   603→  function moveItem(id: string, direction: -1 | 1) {
   604→    setItems((prev) => {
   605→      const target = prev.find((item) => item.id === id);
   606→      if (!target) return prev;
   607→      const siblings = prev
   608→        .filter((item) => item.parent_id === target.parent_id)
   609→        .sort((a, b) => positionOf(a) - positionOf(b));
   610→      const index = siblings.findIndex((item) => item.id === id);
   611→      const other = siblings[index + direction];
   612→      if (!other) return prev;
   613→      return normalizeItems(
   614→        prev.map((item) => {
   615→          if (item.id === id) return { ...item, position: other.position };
   616→          if (item.id === other.id) return { ...item, position: target.position };
   617→          return item;
   618→        })
   619→      );
   620→    });
   621→  }
