   180→function formatDate(iso: string | null | undefined): string {
   181→  if (!iso) return "—";
   182→  const date = new Date(iso);
   183→  return Number.isNaN(date.getTime()) ? "—" : DATE_FORMAT.format(date);
   184→}
   185→
   186→function formatDateTime(iso: string | null | undefined): string {
   187→  if (!iso) return "—";
   188→  const date = new Date(iso);
   189→  return Number.isNaN(date.getTime()) ? "—" : DATETIME_FORMAT.format(date);
   190→}
   191→
   192→/** ISO → valeur datetime-local (heure locale du navigateur). */
   193→function toDatetimeLocal(iso: string | null | undefined): string {
   194→  if (!iso) return "";
   195→  const date = new Date(iso);
   196→  if (Number.isNaN(date.getTime())) return "";
   197→  const pad = (n: number) => String(n).padStart(2, "0");
   198→  return `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(
   199→    date.getDate()
   200→  )}T${pad(date.getHours())}:${pad(date.getMinutes())}`;
   201→}
   202→
   203→/** datetime-local → ISO 8601 (UTC), ou null si vide/invalide. */
   204→function fromDatetimeLocal(value: string): string | null {
   205→  if (!value) return null;
   206→  const date = new Date(value);
   207→  return Number.isNaN(date.getTime()) ? null : date.toISOString();
   208→}

   210→/** Normalise target_paths (chaîne JSON ou tableau) en liste de chemins. */
   211→function parseTargetPaths(raw: string | string[]): string[] {
   212→  let value: unknown = raw;
   213→  if (typeof value === "string") {
   214→    const trimmed = value.trim();
   215→    if (!trimmed) return [];
   216→    try {
   217→      value = JSON.parse(trimmed) as unknown;
   218→    } catch {
   219→      return [];
   220→    }
   221→  }
   222→  if (!Array.isArray(value)) return [];
   223→  return value.filter(
   224→    (entry): entry is string => typeof entry === "string" && entry.trim().length > 0
   225→  );
   226→}
   227→
   228→/** « /a, /b » → ["/a", "/b"]. */
   229→function splitTargetPaths(raw: string): string[] {
   230→  return raw
   231→    .split(",")
   232→    .map((part) => part.trim())
   233→    .filter((part) => part.length > 0);
   234→}

   236→function bannerPeriod(banner: BannerRow): string {
   237→  if (!banner.starts_at && !banner.ends_at) return "Affichage permanent";
   238→  if (banner.starts_at && banner.ends_at) {
   239→    return `Du ${formatDateTime(banner.starts_at)} au ${formatDateTime(banner.ends_at)}`;
   240→  }
   241→  if (banner.starts_at) return `À partir du ${formatDateTime(banner.starts_at)}`;
   242→  return `Jusqu'au ${formatDateTime(banner.ends_at)}`;
   243→}
